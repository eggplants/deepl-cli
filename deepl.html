deepl

deepl-cli

DeepL Translator CLI using playwright-python

Note: This project works without DeepL API key. With DeepL API, use DeepLcom/deepl-python

Install

pip install deepl-cli

Usage

CLI

$ deepl -F en -T ja -s <<<'This tool is useful for me.'
このツールは私にとって役に立ちます。

$ deepl -F ja -T en-US -s <<<'このツールは私にとって便利だ。'
This tool is useful to me.

$ curl https://example.com | grep -oE '>[^<]+<' | tr -d '><' | sed '1,2d' > txt
$ deepl -f txt -F en -T ja

211    async def __translate(self, script: str) -> str:
212        """Throw a request."""
213        async with async_playwright() as p:
214            browser = await self.__get_browser(p)
215            try:
216                page = await self.__open_translator(browser)
217                await self.__input_script(page, script)
218                state = await self.__wait_for_translation(page)
219            finally:
220                await browser.close()
221
222        self.translated_fr_lang = str(state["fr_lang"]).split("-")[0]
223        self.translated_to_lang = str(state["to_lang"]).split("-")[0]

227    async def __open_translator(self, browser: Browser) -> Page:
228        """Open the translator page with the requested language pair selected.
229
230        The language pair is passed through the URL fragment, which DeepL applies on load.
231        That avoids driving the language dropdowns, which are covered by an anti-bot
232        overlay and only respond once the client-side app has hydrated.
233
234        Args:
235            browser (Browser): Browser to open the page in.
236
237        Returns:
238            Page: The loaded translator page.
239
240        Raises:
241            DeepLCLIError: If the page responds with an error status.
242            DeepLCLIPageLoadError: If the translator does not become usable in time.
243        """
244        page = await browser.new_page()
245        page.set_default_timeout(self.timeout)
246        await page.set_viewport_size({"width": 1920, "height": 1080})
247        if self.headless:
248            # Skip what is only needed for a human to look at the page.
249            await page.route(
250                "**/*",
251                lambda route: (
252                    route.abort() if route.request.resource_type in _EXCLUDED_RESOURCES else route.continue_()
253                ),
254            )
255
256        url = f"{TRANSLATOR_URL}#{self.fr_lang}/{self.to_lang}/"
257
258        def is_document_response(resp: Any) -> bool:  # noqa: ANN401
259            # The fragment is never sent to the server, so it is absent from the response URL.
260            return resp.url.split("#")[0] == TRANSLATOR_URL and resp.request.method == "GET"
261
262        async with page.expect_response(is_document_response) as resp_info:
263            await page.goto(url)
264
265        response = await resp_info.value
266
267        if not response.ok:
268            error_text = await page.inner_text("body > main > div > p")
269
270            msg = f"Page loading failed with status code {response.status}: {error_text}"
271            raise DeepLCLIError(msg)
272
273        await self.__wait_for_translator(page)
274
275        return page

277    async def __wait_for_translator(self, page: Page) -> None:
278        """Wait for the source input, answering the cookie dialog if one shows up.
279
280        The dialog is looked for while the app is still loading, which costs nothing.
281        One that has not appeared by the time the input has is left to
282        __wait_for_translation, rather than waited for here at the expense of typing.
283
284        Args:
285            page (Page): Translator page.
286
287        Raises:
288            DeepLCLIPageLoadError: If the translator does not become usable in time.
289        """
290        textbox = page.locator(_SOURCE_TEXTBOX)
291        deadline = time.monotonic() + self.timeout / 1000
292
293        while not await textbox.is_visible():
294            if time.monotonic() > deadline:
295                msg = f"Maybe Time limit exceeded. ({self.timeout} ms)"
296                raise DeepLCLIPageLoadError(msg)
297
298            await self.__accept_cookies(page)
299            await page.wait_for_timeout(_CONSENT_POLL_MS)
300
301        # One last look before handing over to the typing, which happens immediately.
302        await self.__accept_cookies(page)

304    async def __accept_cookies(self, page: Page, *, wait: bool = False) -> bool:
305        """Answer the cookie consent dialog if DeepL is showing one.
306
307        The dialog is laid over the translator, so the source input cannot be
308        clicked until it is gone.
309
310        Args:
311            page (Page): Translator page.
312            wait (bool): Give a dialog time to appear rather than only taking the one
313                that is already there. Only worth doing once something has gone wrong,
314                since idling in front of the translator is what breaks it.
315
316        Returns:
317            bool: Whether a dialog was there and got accepted.
318
319        Raises:
320            DeepLCLIPageLoadError: If the dialog is shown but does not go away.
321        """
322        button = page.locator(_CONSENT_BUTTON).first
323
324        if wait:
325            try:
326                await button.wait_for(state="visible", timeout=_CONSENT_WAIT_MS)
327            except PlaywrightError:
328                return False
329        elif not await button.count() or not await button.is_visible():
330            return False
331
332        try:
333            await self.__click(page, button)
334            await button.wait_for(state="hidden", timeout=_CONSENT_TIMEOUT_MS)
335        except PlaywrightError as e:
336            msg = "The cookie consent dialog did not close"
337            raise DeepLCLIPageLoadError(msg) from e
338
339        return True

341    async def __input_script(self, page: Page, script: str) -> None:
342        """Put the source text into the translator input the way a person would.
343
344        The pointer is moved onto the input and clicks it, then the text is pasted
345        from the clipboard. Setting the value directly is what makes Cloudflare ask
346        for a captcha, so it is only kept as a last resort if the paste is refused
347        (the clipboard is unavailable in some sandboxes).
348
349        This runs as soon as the input appears, and deliberately does not wait around
350        beforehand: a few seconds after load, DeepL stops letting the input take focus
351        at all, and nothing can be entered for the rest of the visit.
352
353        Args:
354            page (Page): Translator page.
355            script (str): Text to translate.
356
357        Raises:
358            DeepLCLIPageLoadError: If the text cannot be entered.
359        """
360        textbox = page.locator(_SOURCE_TEXTBOX)
361        try:
362            entered = await self.__click_and_paste(page, textbox, script)
363
364            if not entered and await self.__accept_cookies(page, wait=True):
365                # A cookie dialog was over the input and took the click; now it is
366                # gone, the same approach can be tried again.
367                entered = await self.__click_and_paste(page, textbox, script)
368
369            if not entered:
370                await self.__retype_script(page, textbox, script)
371                entered = await self.__wait_for_script(textbox, page, script)
372        except PlaywrightError as e:
373            msg = "Unable to enter the source text"
374            raise DeepLCLIPageLoadError(msg) from e
375
376        if not entered:
377            # Reported here rather than left to time out as a translation that never
378            # arrives, which says nothing about the input being the problem.
379            msg = "The source text did not reach the translator input"
380            raise DeepLCLIPageLoadError(msg)

411    @staticmethod
412    async def __click(page: Page, target: Locator) -> None:
413        """Move the pointer onto an element and click it, the way a person would.
414
415        Args:
416            page (Page): Translator page.
417            target (Locator): Element to click.
418        """
419        # Not scrolled into view first: the translator card and the consent dialog are
420        # both within the viewport already, and that call is one DeepL can leave hanging.
421        box = await target.bounding_box()
422
423        if box is None:
424            # No geometry to aim at, so let Playwright place the click itself.
425            await target.click(delay=_CLICK_PRESS_MS)
426        else:
427            x = box["x"] + box["width"] * _CLICK_X_RATIO
428            y = box["y"] + box["height"] * _CLICK_Y_RATIO
429            # A real pointer travels to its target instead of teleporting.
430            await page.mouse.move(x, y, steps=_MOUSE_MOVE_STEPS)
431            await page.wait_for_timeout(_BEFORE_CLICK_MS)
432            await page.mouse.click(x, y, delay=_CLICK_PRESS_MS)
433
434        await page.wait_for_timeout(_AFTER_CLICK_MS)

436    async def __paste_script(self, page: Page, script: str) -> None:
437        """Paste the text into the focused input via the clipboard.
438
439        A headed browser shares the clipboard with the rest of the desktop, so
440        whatever the user had on it is read first and put back afterwards.
441
442        Args:
443            page (Page): Translator page.
444            script (str): Text to translate.
445        """
446        try:
447            await page.context.grant_permissions(_CLIPBOARD_PERMISSIONS)
448            clipboard: str | None = await page.evaluate(_READ_CLIPBOARD_JS)
449            await page.evaluate(_WRITE_CLIPBOARD_JS, script)
450        except PlaywrightError:
451            # No clipboard access; __input_script falls back to typing the text.
452            return
453
454        try:
455            await page.keyboard.press(_PASTE_SHORTCUT)
456            await page.wait_for_timeout(_AFTER_CLICK_MS)
457        finally:
458            await self.__restore_clipboard(page, clipboard)

460    @staticmethod
461    async def __restore_clipboard(page: Page, clipboard: str | None) -> None:
462        """Put the user's own clipboard contents back.
463
464        Only text can be restored: anything else was already unreadable, and is
465        left replaced by the source text.
466
467        Args:
468            page (Page): Translator page.
469            clipboard (str | None): What the clipboard held before pasting.
470        """
471        if not clipboard:
472            return
473
474        # Losing the clipboard is not worth failing a translation over.
475        with suppress(PlaywrightError):
476            await page.evaluate(_WRITE_CLIPBOARD_JS, clipboard)

478    @staticmethod
479    async def __wait_for_script(textbox: Locator, page: Page, script: str) -> bool:
480        """Wait for the input to hold the text, ignoring how it wraps lines.
481
482        The editor renders what it is given a moment later, so a single check right
483        after the paste can miss text that did arrive.
484
485        Args:
486            textbox (Locator): The source input.
487            page (Page): Translator page.
488            script (str): Text that should be in the input.
489
490        Returns:
491            bool: Whether the text is in the input.
492        """
493        wanted = _WHITESPACE.sub(" ", script).strip()
494
495        for _ in range(_INPUT_CHECK_ATTEMPTS):
496            entered = await textbox.inner_text()
497            if _WHITESPACE.sub(" ", entered).strip() == wanted:
498                return True
499            await page.wait_for_timeout(_INPUT_CHECK_INTERVAL_MS)
500
501        return False

503    @staticmethod
504    async def __retype_script(page: Page, textbox: Locator, script: str) -> None:
505        """Replace whatever the paste left behind with the text itself.
506
507        `insert_text` reaches the page as one `insertText` event, the same way text
508        committed from an IME does, so it does not look like a filled-in form either.
509
510        Args:
511            page (Page): Translator page.
512            textbox (Locator): The source input.
513            script (str): Text to translate.
514        """
515        await textbox.focus()
516        # A partial paste would otherwise be left in front of the text.
517        await page.keyboard.press(_SELECT_ALL_SHORTCUT)
518        await page.keyboard.press("Delete")
519        await page.keyboard.insert_text(script)

521    async def __wait_for_translation(self, page: Page) -> _PageState:
522        """Wait until the translation is complete and stable.
523
524        DeepL streams the output sentence by sentence, so a non-empty output is not
525        necessarily the final one. Poll until the same output is seen twice in a row.
526
527        A cookie dialog is answered here if one turns up, which is where it usually
528        does: the consent script starts up later than the translator. Doing it during
529        a wait that was happening anyway keeps it off the critical path, and text
530        already in the input goes on being translated meanwhile.
531
532        Args:
533            page (Page): Translator page.
534
535        Returns:
536            _PageState: The settled state of the translator widget.
537
538        Raises:
539            DeepLCLIPageLoadError: If the translation does not settle in time.
540        """
541        deadline = time.monotonic() + self.timeout / 1000
542        state = await self.__read_state(page)
543        previous: _PageState | None = None
544        dialog_pending = True
545
546        while time.monotonic() < deadline:
547            if dialog_pending:
548                try:
549                    dialog_pending = not await self.__accept_cookies(page)
550                except DeepLCLIPageLoadError:
551                    # A dialog that will not close is no reason to throw away a
552                    # translation that is arriving regardless, so it is left alone.
553                    dialog_pending = False
554
555            if self.__has_output(state):
556                settled = previous is not None and previous["text"] == state["text"]
557
558                # A challenged page will never get as far as the requested pair, so the
559                # translation on screen is taken as final rather than waited out.
560                if settled and (self.__is_translated(state) or state["challenged"]):
561                    return state
562
563                previous = state
564            else:
565                previous = None
566
567            await page.wait_for_timeout(_POLL_INTERVAL_MS)
568            state = await self.__read_state(page)
569
570        raise DeepLCLIPageLoadError(self.__describe_timeout(state))

572    async def __read_state(self, page: Page) -> _PageState:
573        """Read the current languages and output text from the translator widget."""
574        try:
575            state: _PageState = await page.evaluate(_READ_STATE_JS)

577            msg = "Unable to read the translator state"

582    def __is_translated(self, state: _PageState) -> bool:
583        """Check whether the state holds a complete translation of the requested pair."""
584        return (
585            self.__lang_applied(state["fr_lang"], self.fr_lang)
586            and self.__lang_applied(state["to_lang"], self.to_lang)
587            and self.__has_output(state)
588        )
589
590    @staticmethod
591    def __has_output(state: _PageState) -> bool:
592        """Check whether the output holds a translation that is no longer mid-sentence."""
593        text = state["text"]
594
595        # An input the user cannot see anything in must not read as a finished result,
596        # so whitespace on its own does not count as output.
597        return bool(text) and bool(text.strip()) and _PENDING_MARKER not in text
598
599    @staticmethod
600    def __lang_applied(actual: str | None, expected: str) -> bool:
601        """Check whether DeepL selected the expected language (it reports e.g. `en-US`)."""
602        return actual is not None and actual.lower() == expected.lower()
603
604    def __describe_timeout(self, state: _PageState) -> str:
605        """Explain why waiting for the translation timed out."""
606        if state["challenged"]:
607            return (
608                "DeepL is checking whether the connection is secure and nothing was "
609                f"translated behind that check. ({self.timeout} ms)"
610            )
611
612        if not self.__lang_applied(state["fr_lang"], self.fr_lang) or not self.__lang_applied(
613            state["to_lang"],
614            self.to_lang,
615        ):
616            return (
617                f"DeepL did not select the requested language pair "
618                f"(wanted {self.fr_lang!r} -> {self.to_lang!r}, "
619                f"got {state['fr_lang']!r} -> {state['to_lang']!r}). ({self.timeout} ms)"
620            )
621
622        return f"Time limit exceeded. ({self.timeout} ms)"
623
624    def __sanitize_script(self, script: str) -> str:
625        """Check command line args and stdin."""
626        script = script.rstrip("\n")
627
628        if self.max_length is not None and len(script) > self.max_length:
629            msg = f"Limit of script is less than {self.max_length} chars (Now: {len(script)} chars)"
630            raise DeepLCLIError(msg)
631
632        if len(script) <= 0:
633            msg = "Script seems to be empty."
634            raise DeepLCLIError(msg)
635
636        return script

638    async def __get_browser(self, p: Playwright) -> Browser:
639        """Launch browser executable and get playwright browser object."""
640        install([p.chromium], with_deps=True)
641
642        args = [
643            "--no-sandbox",
644            "--disable-dev-shm-usage",
645            "--window-size=1920,1080",
646        ]
647        if self.headless:
648            # These make the headless browser cheaper to start, but a headed
649            # Chromium does not survive them.
650            args += ["--disable-gpu", "--no-zygote"]
651            if os.name != "nt":
652                args.append("--single-process")
653
654        return await p.chromium.launch(
655            headless=self.headless,
656            args=args,
657            proxy=self.proxy,
658        )
